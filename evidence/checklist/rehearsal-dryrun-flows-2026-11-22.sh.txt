export QS_RIGHTS_CONTRACT=$C QS_PILOT_ACCOUNTS=CBK3H55VKCKZ4LRIZSETP3ZYP4OI4IIBN3SCCCFNYI5WDFC45PSWWEEL
declare -A WHO=([issue-a]=qs-issuer [issue-b]=qs-issuer [sale-request]=qs-buyer [sale-offer]=qs-buyer [sale-accept]=qs-owner [rental-offer]=qs-renter)
n=0
for step in ${STEPS:-issue-a issue-b sale-request sale-offer sale-accept rental-offer}; do
  f=$(ls "$OUT"/*-"$step".xdr 2>/dev/null || true); [ -n "$f" ] || f="$OUT/$(printf %02d $(( $(ls "$OUT"/*.xdr 2>/dev/null | wc -l) + 1 )))-$step.xdr"
  npx tsx scripts/pilot-envelope.ts "$step" --network testnet --contract "$C" --out "$OUT"
  printf 'SEND\n' | scripts/mainnet-sign-submit.sh "$f" "${WHO[$step]}" --network testnet
done
npx tsx scripts/pilot-envelope.ts status --network testnet --contract "$C" --out "$OUT"
